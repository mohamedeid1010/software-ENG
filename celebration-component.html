الخطوة 3: كود الـ JavaScript انسخ وضعه قبل وسم الإغلاق body في script

<script>
  // تفاعل النجوم الطائرة عند تمرير الفأرة فوق الزر
  const finishBtn = document.getElementById('finish-study-btn');
  const finishBtnWrapper = document.querySelector('.finish-btn-wrapper');
  if (finishBtn && finishBtnWrapper) {

    class Sparkle {
      constructor(x, y, color, isExplosion = true) {
        this.x = x;
        this.y = y;
        this.size = Math.random() * (isExplosion ? 5 : 2.5) + 1.2;
        this.speedX = isExplosion ? (Math.random() - 0.5) * 16 : (Math.random() - 0.5) * 2.5;
        this.speedY = isExplosion ? (Math.random() - 0.75) * 16 - 3 : -Math.random() * 1.5 - 0.5;
        this.gravity = isExplosion ? 0.16 : 0.01;
        this.color = color;
        this.alpha = 1;
        this.decay = isExplosion ? (Math.random() * 0.015 + 0.007) : 0.005;
      }
      update() {
        this.x += this.speedX;
        this.y += this.speedY;
        this.speedY += this.gravity;
        this.alpha -= this.decay;
      }
      draw() {
        ctx.save();
        ctx.globalAlpha = this.alpha;
        ctx.beginPath();
        if (this.size > 3) {
          ctx.shadowBlur = 10;
          ctx.shadowColor = this.color;

          let cy = this.y;

          ctx.moveTo(cx, cy - outerRadius);
          for (let i = 0; i < spikes; i++) {
            cx = this.x + Math.cos(rot) * outerRadius;
            cy = this.y + Math.sin(rot) * outerRadius;

            cy = this.y + Math.sin(rot) * innerRadius;

          ctx.closePath();

        }

        ctx.fill();
        ctx.restore();
      }

      overlay.style.display = 'flex';
      overlay.offsetHeight;
      overlay.style.opacity = '1';
      
      burst(50, canvas.height - 50, 80);
      burst(canvas.width - 50, canvas.height - 50, 80);
      burst(canvas.width / 2, canvas.height / 3, 70);
    }, 500);
    
    let animationFrameId;
    let keepSpawning = true;
    
    function animate() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      
      for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.update();
        p.draw();
        if (p.alpha <= 0 || (p instanceof Confetti && p.y > canvas.height + 50)) {
          particles.splice(i, 1);
        }
      }
      
      if (keepSpawning && Math.random() < 0.15) {
        particles.push(new Sparkle(
          Math.random() * canvas.width,
          canvas.height + 10,
          colors[Math.floor(Math.random() * colors.length)],
          false
        ));
      }
      
      animationFrameId = requestAnimationFrame(animate);
    }
    
    animate();
    
    function handleMouseMove(e) {
      if (Math.random() > 0.3) return;
      particles.push(new Sparkle(
        e.clientX,
        e.clientY,
        colors[Math.floor(Math.random() * colors.length)],
        false
      ));
    }
    overlay.addEventListener('mousemove', handleMouseMove);
    
    const card1 = document.getElementById('card-step-1');
    const card2 = document.getElementById('card-step-2');
    const card3 = document.getElementById('card-step-3');
    
    card1.classList.add('active');
    card1.classList.remove('exit-left', 'enter-right');
    card2.classList.remove('active', 'exit-left', 'enter-right');
    card3.classList.remove('active', 'exit-left', 'enter-right');
    
    const nextBtn1 = card1.querySelector('.next-step-btn');
    const nextBtn2 = card2.querySelector('.next-step-btn');
    
    function transitionToCard(currentCard, nextCard) {
      currentCard.classList.remove('active');
      currentCard.classList.add('exit-left');
      
      nextCard.classList.remove('exit-left');
      nextCard.classList.add('enter-right');
      
      nextCard.offsetHeight;
      
      nextCard.classList.remove('enter-right');
      nextCard.classList.add('active');
    }

    nextBtn1.addEventListener('click', function() {
      transitionToCard(card1, card2);
      burst(canvas.width / 2, canvas.height / 2 - 80, 50);
    });
    
    nextBtn2.addEventListener('click', function() {
      transitionToCard(card2, card3);
      burst(canvas.width / 3, canvas.height / 2 - 80, 45);
      burst(canvas.width * 2 / 3, canvas.height / 2 - 80, 45);
    });
    
    document.getElementById('close-overlay-btn').addEventListener('click', function() {
      overlay.style.opacity = '0';
      ripple.style.transform = 'scale(0)';
      ripple.style.opacity = '0';
      
      overlay.removeEventListener('mousemove', handleMouseMove);
      
      setTimeout(() => {
        overlay.style.display = 'none';
        overlay.style.opacity = '1';
        ripple.style.display = 'none';
        
        card1.classList.remove('active', 'exit-left', 'enter-right');
        card2.classList.remove('active', 'exit-left', 'enter-right');
        card3.classList.remove('active', 'exit-left', 'enter-right');
        
        keepSpawning = false;
        cancelAnimationFrame(animationFrameId);
        window.removeEventListener('resize', resizeCanvas);
      }, 600);
    }, { once: true });
  });
</script>
