الخطوة 3: كود الـ JavaScript انسخ وضعه قبل وسم الإغلاق body في script

هذا الكود يتحكم في تفاعل الزر، تشغيل الأنيميشن والـ Canvas، والتنقل بين بطاقات الاحتفال الثلاثة.

<script>
  document.getElementById('finish-study-btn').addEventListener('click', function() {
    const overlay = document.getElementById('celebration-overlay');
    const canvas = document.getElementById('sparkles-canvas');
    const ctx = canvas.getContext('2d');
    
    overlay.style.display = 'flex';
    
    // Canvas sizing
    function resizeCanvas() {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
    }
    resizeCanvas();
    window.addEventListener('resize', resizeCanvas);
    
    // Particle constructor / class
    class Particle {
      constructor(x, y, color, isExplosion = true) {
        this.x = x;
        this.y = y;
        this.size = Math.random() * (isExplosion ? 4 : 2) + 1;
        this.speedX = isExplosion ? (Math.random() - 0.5) * 14 : (Math.random() - 0.5) * 2;
        this.speedY = isExplosion ? (Math.random() - 0.7) * 14 - 3 : -Math.random() * 1.5 - 0.5;
        this.gravity = isExplosion ? 0.18 : 0;
        this.color = color;
        this.alpha = 1;
        this.decay = isExplosion ? (Math.random() * 0.015 + 0.005) : 0.005;
      }
      update() {
        this.x += this.speedX;
        this.y += this.speedY;
        this.speedY += this.gravity;
        this.alpha -= this.decay;
      }
      draw() {
        ctx.save();
        ctx.globalAlpha = this.alpha;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        ctx.fillStyle = this.color;
        if (this.size > 2) {
          ctx.shadowBlur = 8;
          ctx.shadowColor = this.color;
        }
        ctx.fill();
        ctx.restore();
      }
    }
    
    let particles = [];
    const colors = ['#e8b85a', '#f4d28a', '#ffffff', '#e9e6df', '#ffd700'];
    
    // Spawn initial explosion from corners and center
    function burst() {
      // Left bottom
      for (let i = 0; i < 60; i++) {
        particles.push(new Particle(0, canvas.height, colors[Math.floor(Math.random() * colors.length)]));
      }
      // Right bottom
      for (let i = 0; i < 60; i++) {
        particles.push(new Particle(canvas.width, canvas.height, colors[Math.floor(Math.random() * colors.length)]));
      }
      // Center
      for (let i = 0; i < 50; i++) {
        particles.push(new Particle(canvas.width / 2, canvas.height / 2 + 100, colors[Math.floor(Math.random() * colors.length)]));
      }
    }
    
    burst();
    
    let animationFrameId;
    let keepSpawning = true;
    
    function animate() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      
      // Update and draw particles
      for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.update();
        p.draw();
        if (p.alpha <= 0) {
          particles.splice(i, 1);
        }
      }
      
      // Periodically spawn floating background stars
      if (keepSpawning && Math.random() < 0.2) {
        particles.push(new Particle(
          Math.random() * canvas.width,
          canvas.height + 10,
          colors[Math.floor(Math.random() * colors.length)],
          false
        ));
      }
      
      animationFrameId = requestAnimationFrame(animate);
    }
    
    animate();
    
    // Mouse trailing interaction
    function handleMouseMove(e) {
      for (let i = 0; i < 2; i++) {
        particles.push(new Particle(
          e.clientX,
          e.clientY,
          colors[Math.floor(Math.random() * colors.length)],
          false
        ));
      }
    }
    overlay.addEventListener('mousemove', handleMouseMove);
    
    // Sequencing the cards
    const card1 = document.getElementById('card-step-1');
    const card2 = document.getElementById('card-step-2');
    const card3 = document.getElementById('card-step-3');
    
    // Reset states
    card1.classList.remove('active');
    card2.classList.remove('active');
    card3.classList.remove('active');
    
    // Step 1
    setTimeout(() => {
      card1.classList.add('active');
    }, 200);
    
    // Step 2
    let step2Timeout = setTimeout(() => {
      card1.classList.remove('active');
      setTimeout(() => {
        card2.classList.add('active');
        // Trigger another light burst
        for (let i = 0; i < 40; i++) {
          particles.push(new Particle(canvas.width / 2, canvas.height / 2, colors[Math.floor(Math.random() * colors.length)]));
        }
      }, 800);
    }, 4500); // Show card1 for 4.5 seconds
    
    // Step 3
    let step3Timeout = setTimeout(() => {
      card2.classList.remove('active');
      setTimeout(() => {
        card3.classList.add('active');
        // Final gold burst
        burst();
      }, 800);
    }, 9500); // Show card2 for 5 seconds
    
    // Close handler
    document.getElementById('close-overlay-btn').addEventListener('click', function() {
      overlay.style.opacity = '0';
      overlay.style.transition = 'opacity 0.5s ease';
      
      overlay.removeEventListener('mousemove', handleMouseMove);
      clearTimeout(step2Timeout);
      clearTimeout(step3Timeout);
      
      setTimeout(() => {
        overlay.style.display = 'none';
        overlay.style.opacity = '1';
        overlay.style.transition = '';
        card1.classList.remove('active');
        card2.classList.remove('active');
        card3.classList.remove('active');
        keepSpawning = false;
        cancelAnimationFrame(animationFrameId);
        window.removeEventListener('resize', resizeCanvas);
      }, 500);
    }, { once: true });
  });
</script>
